include(FetchContent)
# Dedicated-server IP transport needs neither Steam services nor P2P ICE.
set(BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_ICE OFF CACHE BOOL "" FORCE)
set(USE_STEAMWEBRTC OFF CACHE BOOL "" FORCE)
FetchContent_Declare(gns GIT_REPOSITORY https://github.com/ValveSoftware/GameNetworkingSockets.git
  GIT_TAG 2cb93a06350bb065db53abdb0d87cf297e0bfd34 GIT_SUBMODULES "")
FetchContent_MakeAvailable(gns)
# Upstream disables RTTI. Keep it on in sanitizer builds so project UBSan vptr
# checks can inspect the derived GNS objects across the library boundary.
if(OFS_SANITIZE AND NOT MSVC)
  target_compile_options(GameNetworkingSockets_s PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-frtti>)
endif()
add_library(ofs_net STATIC src/bot_ai.cpp src/combat.cpp src/weapons.cpp src/weapons_protocol.cpp src/protocol.cpp src/replication.cpp src/transport.cpp src/world.cpp src/server.cpp src/client.cpp src/discovery.cpp)
target_include_directories(ofs_net PUBLIC include)
target_link_libraries(ofs_net PUBLIC ofs_core PRIVATE GameNetworkingSockets::static)
if(WIN32)
  # LAN discovery uses plain UDP sockets beside the transport.
  target_link_libraries(ofs_net PRIVATE ws2_32)
endif()
get_target_property(gns_headers GameNetworkingSockets_s INTERFACE_INCLUDE_DIRECTORIES)
target_include_directories(ofs_net SYSTEM PRIVATE ${gns_headers})
ofs_target_defaults(ofs_net)
add_executable(ofs_server src/server_main.cpp)
install(TARGETS ofs_server RUNTIME_DEPENDENCY_SET ofs_runtime RUNTIME DESTINATION .)
target_link_libraries(ofs_server PRIVATE ofs_net)
target_include_directories(ofs_server PRIVATE "${CMAKE_BINARY_DIR}/generated")
ofs_target_defaults(ofs_server)
add_executable(ofs_bot src/bot_main.cpp)
target_link_libraries(ofs_bot PRIVATE ofs_net)
ofs_target_defaults(ofs_bot)
add_executable(ofs_net_benchmark src/benchmark.cpp)
target_link_libraries(ofs_net_benchmark PRIVATE ofs_net)
ofs_target_defaults(ofs_net_benchmark)
add_executable(ofs_net_load src/load.cpp)
target_link_libraries(ofs_net_load PRIVATE ofs_net)
ofs_target_defaults(ofs_net_load)
add_executable(ofs_replication_benchmark src/replication_benchmark.cpp)
target_link_libraries(ofs_replication_benchmark PRIVATE ofs_net)
ofs_target_defaults(ofs_replication_benchmark)
